cmake_minimum_required(VERSION 3.25)
project(sr_xdk_constant_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(SR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(SR_SDK "${SR_ROOT}/.tools/rexglue-sdk/win-amd64" CACHE PATH "ReXGlue SDK")
function(extract_constant_reference file symbol reference)
  file(READ "${SR_ROOT}/port/generated/default/${file}" source)
  string(FIND "${source}" "DEFINE_REX_FUNC(${symbol})" begin)
  if(begin LESS 0)
    message(FATAL_ERROR "Missing generated constant function ${symbol}")
  endif()
  string(SUBSTRING "${source}" ${begin} -1 rest)
  string(FIND "${rest}" "\nDEFINE_REX_FUNC(" end)
  if(end LESS 0)
    message(FATAL_ERROR "Missing function boundary after ${symbol}")
  endif()
  string(SUBSTRING "${rest}" 0 ${end} function)
  string(REPLACE "DEFINE_REX_FUNC(${symbol})" "REX_FUNC(${reference})" function "${function}")
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/constant_references.cpp" "${function}\n")
endfunction()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/constant_references.cpp" "#include \"superman_returns_pch.h\"\n")
extract_constant_reference(superman_returns_recomp.9.cpp sub_820F4920 reference_vs)
extract_constant_reference(superman_returns_recomp.75.cpp sub_820F4A30 reference_ps)
add_executable(sr_xdk_constant_tests test_constant_copy.cpp "${CMAKE_CURRENT_BINARY_DIR}/constant_references.cpp")
target_include_directories(sr_xdk_constant_tests PRIVATE "${SR_SDK}/include" "${SR_ROOT}/port/generated/default")
target_compile_options(sr_xdk_constant_tests PRIVATE -mssse3 -msse4.1)
target_compile_definitions(sr_xdk_constant_tests PRIVATE SPDLOG_FMT_EXTERNAL)
enable_testing()
add_test(NAME sr_xdk_constant_tests COMMAND sr_xdk_constant_tests)
