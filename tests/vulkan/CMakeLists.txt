cmake_minimum_required(VERSION 3.25)
project(sr_vulkan_foundation LANGUAGES CXX)
enable_testing()
add_subdirectory(../../port/src/graphics/vulkan vulkan)
add_executable(sr_vulkan_tests main.cpp test_policy.cpp test_context.cpp test_frame_loop.cpp test_swapchain.cpp test_triangle.cpp test_smoke_options.cpp test_device_requirements.cpp test_descriptors.cpp test_resources.cpp test_render_targets.cpp test_image_state.cpp test_shader_service.cpp test_game_pipeline.cpp test_resolve.cpp)
target_include_directories(sr_vulkan_tests PRIVATE ../native)
target_link_libraries(sr_vulkan_tests PRIVATE sr_vulkan_core)
target_sources(sr_vulkan_tests PRIVATE test_composition.cpp test_state_shadow.cpp)
target_sources(sr_vulkan_tests PRIVATE test_global_descriptors.cpp)
target_sources(sr_vulkan_tests PRIVATE test_global_draw_bindings.cpp)
add_test(NAME sr_vulkan_tests COMMAND sr_vulkan_tests)

if(WIN32)
  add_executable(sr_vulkan_shader_process_test shader_process_integration.cpp ../../port/src/graphics/vulkan/platform/shader_process.cpp)
  target_link_libraries(sr_vulkan_shader_process_test PRIVATE sr_vulkan_core)
  target_compile_definitions(sr_vulkan_shader_process_test PRIVATE NOMINMAX)
  add_executable(sr_vulkan_resources_test resources_integration.cpp ../../port/src/graphics/vulkan/platform/win32_surface.cpp)
  add_library(sr_vulkan_capture_test_support STATIC
      ../../port/src/graphics/guest/render_packet.cpp
      ../../port/src/graphics/guest/draw_state.cpp
      ../../port/src/graphics/guest/render_state.cpp
      ../../port/src/graphics/guest/constant_snapshot.cpp
      ../../port/src/native_renderer/pm4_mirror.cpp)
  target_link_libraries(sr_vulkan_capture_test_support PUBLIC sr_vulkan_core)
  target_link_libraries(sr_vulkan_resources_test PRIVATE sr_vulkan_capture_test_support user32)
  target_compile_definitions(sr_vulkan_resources_test PRIVATE VK_USE_PLATFORM_WIN32_KHR NOMINMAX)
  set(target_shader_dir "${CMAKE_CURRENT_BINARY_DIR}/target-shaders")
  file(MAKE_DIRECTORY "${target_shader_dir}")
  foreach(stage IN ITEMS vs ps)
    if(stage STREQUAL "vs")
      set(entry VSMain)
    else()
      set(entry PSMain)
    endif()
    add_custom_command(OUTPUT "${target_shader_dir}/${stage}.spv"
      COMMAND "${SR_VULKAN_DXC}" -spirv -fspv-target-env=vulkan1.1 -T ${stage}_6_0 -E ${entry} -Fo "${target_shader_dir}/${stage}.spv" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/target_tests.hlsl"
      DEPENDS shaders/target_tests.hlsl VERBATIM)
    target_sources(sr_vulkan_resources_test PRIVATE "${target_shader_dir}/${stage}.spv")
  endforeach()
  target_compile_definitions(sr_vulkan_resources_test PRIVATE SR_TARGET_SHADER_DIR="${target_shader_dir}")
  add_custom_command(OUTPUT "${target_shader_dir}/resolve_sample.spv"
    COMMAND "${SR_VULKAN_DXC}" -spirv -fspv-target-env=vulkan1.1 -T ps_6_0 -E PSMain -Fo "${target_shader_dir}/resolve_sample.spv" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/resolve_sample.hlsl"
    DEPENDS shaders/resolve_sample.hlsl VERBATIM)
  target_sources(sr_vulkan_resources_test PRIVATE "${target_shader_dir}/resolve_sample.spv")
  foreach(entry IN ITEMS PS0 PS1 PSDynamic)
    add_custom_command(OUTPUT "${target_shader_dir}/constant_mask_${entry}.spv"
      COMMAND "${SR_VULKAN_DXC}" -spirv -fspv-target-env=vulkan1.1 -T ps_6_0 -E ${entry} -Fo "${target_shader_dir}/constant_mask_${entry}.spv" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/constant_mask.hlsl"
      DEPENDS shaders/constant_mask.hlsl VERBATIM)
    target_sources(sr_vulkan_resources_test PRIVATE "${target_shader_dir}/constant_mask_${entry}.spv")
  endforeach()
  option(SR_VULKAN_GPU_TESTS "Run hardware-dependent Vulkan tests in CTest" OFF)
  if(SR_VULKAN_GPU_TESTS)
    foreach(mode IN ITEMS 0 1)
      foreach(path IN ITEMS texture-updates texture-updates-separated)
        add_test(NAME sr_vulkan_${path}_${mode}_gpu COMMAND sr_vulkan_resources_test --${path})
        set_tests_properties(sr_vulkan_${path}_${mode}_gpu PROPERTIES
          ENVIRONMENT "SR_NATIVE_TEXTURE_UPDATES=${mode}" TIMEOUT 60)
      endforeach()
    endforeach()
    foreach(mask IN ITEMS 0 1)
      add_test(NAME sr_vulkan_constant_mask_${mask}_gpu COMMAND sr_vulkan_resources_test --constant-mask)
      set_tests_properties(sr_vulkan_constant_mask_${mask}_gpu PROPERTIES
        ENVIRONMENT "SR_VULKAN_CONSTANT_BANK_CACHE=1;SR_NATIVE_USED_CONSTANT_MASK=${mask}"
        TIMEOUT 60)
    endforeach()
  endif()
  set(SR_VULKAN_GAME_COMMON "${CMAKE_CURRENT_SOURCE_DIR}/../../build/vulkan-m2/emitter-tree/src/XenosRecomp/shader_common.h" CACHE FILEPATH "Patched project shader common header")
  if(EXISTS "${SR_VULKAN_GAME_COMMON}")
    get_filename_component(contract_include "${SR_VULKAN_GAME_COMMON}" DIRECTORY)
    set(contract_dir "${CMAKE_CURRENT_BINARY_DIR}/contract-shaders")
    file(MAKE_DIRECTORY "${contract_dir}")
    foreach(stage IN ITEMS vs ps)
      if(stage STREQUAL "vs")
        set(entry VSMain)
        set(yflag -fvk-invert-y)
      else()
        set(entry PSMain)
        set(yflag "")
      endif()
      add_custom_command(OUTPUT "${contract_dir}/${stage}.spv"
        COMMAND "${SR_VULKAN_DXC}" -spirv -fspv-target-env=vulkan1.1 -D SR_VULKAN_BUFFERS=1 -HV 2021 -T ${stage}_6_0 -E ${entry} ${yflag} -I "${contract_include}" -Fo "${contract_dir}/${stage}.spv" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/guest_contract.hlsl"
        DEPENDS shaders/guest_contract.hlsl ../shaders/fixtures/contract.hlsl "${SR_VULKAN_GAME_COMMON}" VERBATIM)
    endforeach()
    add_executable(sr_vulkan_contract_test shader_contract_integration.cpp ../../port/src/graphics/vulkan/platform/win32_surface.cpp ../../port/src/graphics/guest/draw_state.cpp "${contract_dir}/vs.spv" "${contract_dir}/ps.spv")
    target_link_libraries(sr_vulkan_contract_test PRIVATE sr_vulkan_core user32)
    target_compile_definitions(sr_vulkan_contract_test PRIVATE VK_USE_PLATFORM_WIN32_KHR NOMINMAX)
    if(SR_VULKAN_GPU_TESTS)
      add_test(NAME sr_vulkan_contract_gpu COMMAND sr_vulkan_contract_test)
      add_test(NAME sr_vulkan_texture_mask_gpu COMMAND sr_vulkan_resources_test --texture-mask)
      add_test(NAME sr_vulkan_texture_storage_gpu COMMAND sr_vulkan_resources_test --texture-storage)
      add_test(NAME sr_vulkan_texture_storage_merged_gpu COMMAND sr_vulkan_resources_test --texture-storage-merged)
    endif()
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      set(global_dir "${CMAKE_CURRENT_BINARY_DIR}/global-shaders")
      file(MAKE_DIRECTORY "${global_dir}")
      add_custom_command(OUTPUT "${global_dir}/global_common.h"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_global_common.py" "${SR_VULKAN_GAME_COMMON}" "${global_dir}/global_common.h"
        DEPENDS make_global_common.py ../../tools/shaders/vulkan_global_contract.py "${SR_VULKAN_GAME_COMMON}" VERBATIM)
      add_custom_command(OUTPUT "${global_dir}/texture_storage_ps.spv"
        COMMAND "${SR_VULKAN_DXC}" -spirv -fspv-target-env=vulkan1.1 -fspv-extension=SPV_EXT_descriptor_indexing -T ps_6_0 -E PSMain -I "${global_dir}" -I "${contract_include}" -Fo "${global_dir}/texture_storage_ps.spv" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/texture_storage_global.hlsl"
        DEPENDS shaders/texture_storage_global.hlsl "${global_dir}/global_common.h" "${SR_VULKAN_GAME_COMMON}" VERBATIM)
      list(APPEND global_shader_outputs "${global_dir}/texture_storage_ps.spv")
      foreach(mode IN ITEMS global legacy)
        foreach(stage IN ITEMS vs ps)
          if(stage STREQUAL "vs")
            set(global_entry VSMain)
          else()
            set(global_entry PSMain)
          endif()
          if(mode STREQUAL "global")
            set(global_define -D SR_GLOBAL_TEST=1)
          else()
            set(global_define "")
          endif()
          add_custom_command(OUTPUT "${global_dir}/${mode}_${stage}.spv"
            COMMAND "${SR_VULKAN_DXC}" -spirv -fspv-target-env=vulkan1.1 -fspv-extension=SPV_EXT_descriptor_indexing -T ${stage}_6_0 -E ${global_entry} ${global_define} -I "${global_dir}" -I "${contract_include}" -Fo "${global_dir}/${mode}_${stage}.spv" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/global_contract.hlsl"
            DEPENDS shaders/global_contract.hlsl "${global_dir}/global_common.h" "${SR_VULKAN_GAME_COMMON}" VERBATIM)
          list(APPEND global_shader_outputs "${global_dir}/${mode}_${stage}.spv")
        endforeach()
      endforeach()
      foreach(mode IN ITEMS sample2d sample3d samplecube)
        if(mode STREQUAL "sample2d")
          set(texture_mode 1)
        elseif(mode STREQUAL "sample3d")
          set(texture_mode 2)
        else()
          set(texture_mode 3)
        endif()
        add_custom_command(OUTPUT "${global_dir}/global_${mode}_ps.spv"
          COMMAND "${SR_VULKAN_DXC}" -spirv -fspv-target-env=vulkan1.1 -fspv-extension=SPV_EXT_descriptor_indexing -T ps_6_0 -E PSMain -D SR_GLOBAL_TEST=1 -D SR_GLOBAL_TEXTURE_MODE=${texture_mode} -I "${global_dir}" -I "${contract_include}" -Fo "${global_dir}/global_${mode}_ps.spv" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/global_contract.hlsl"
          DEPENDS shaders/global_contract.hlsl "${global_dir}/global_common.h" "${SR_VULKAN_GAME_COMMON}" VERBATIM)
        list(APPEND global_shader_outputs "${global_dir}/global_${mode}_ps.spv")
      endforeach()
      foreach(mode IN ITEMS global legacy)
        if(mode STREQUAL "global")
          set(global_define -D SR_GLOBAL_TEST=1)
        else()
          set(global_define "")
        endif()
        add_custom_command(OUTPUT "${global_dir}/${mode}_ia_vs.spv"
          COMMAND "${SR_VULKAN_DXC}" -spirv -fspv-target-env=vulkan1.1 -fspv-extension=SPV_EXT_descriptor_indexing -T vs_6_0 -E VSMain -D SR_GLOBAL_IA=1 ${global_define} -I "${global_dir}" -I "${contract_include}" -Fo "${global_dir}/${mode}_ia_vs.spv" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/global_contract.hlsl"
          DEPENDS shaders/global_contract.hlsl "${global_dir}/global_common.h" "${SR_VULKAN_GAME_COMMON}" VERBATIM)
        list(APPEND global_shader_outputs "${global_dir}/${mode}_ia_vs.spv")
      endforeach()
      add_executable(sr_vulkan_global_test global_descriptors_integration.cpp ../../port/src/graphics/vulkan/platform/win32_surface.cpp ${global_shader_outputs})
      target_link_libraries(sr_vulkan_global_test PRIVATE sr_vulkan_capture_test_support user32)
      target_compile_definitions(sr_vulkan_global_test PRIVATE VK_USE_PLATFORM_WIN32_KHR NOMINMAX SR_TARGET_SHADER_DIR="${target_shader_dir}" SR_GLOBAL_SHADER_DIR="${global_dir}")
      if(SR_VULKAN_GPU_TESTS)
        add_test(NAME sr_vulkan_global_gpu COMMAND sr_vulkan_global_test)
        add_test(NAME sr_vulkan_texture_storage_global_gpu COMMAND sr_vulkan_global_test --texture-storage)
      endif()
    endif()
  endif()
endif()
