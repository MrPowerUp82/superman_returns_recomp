cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sr_xma_tests LANGUAGES CXX)
  find_package(rexglue 0.10.0 REQUIRED CONFIG)
endif()
add_executable(sr_xma_work_tests test_work.cpp)
target_compile_features(sr_xma_work_tests PRIVATE cxx_std_23)
target_link_libraries(sr_xma_work_tests PRIVATE rex::runtime)
add_custom_command(TARGET sr_xma_work_tests POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:rex::runtime> $<TARGET_FILE_DIR:sr_xma_work_tests>)
enable_testing()
add_test(NAME sr_xma_work COMMAND sr_xma_work_tests)
add_test(NAME sr_xma_stall COMMAND sr_xma_work_tests --stall)
set_tests_properties(sr_xma_work sr_xma_stall PROPERTIES TIMEOUT 15)
