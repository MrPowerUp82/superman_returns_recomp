cmake_minimum_required(VERSION 3.24)
project(sr_clip_motion_replay LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Requires Windows D3D12 and an actual RTX 2060 (no WARP/skips)")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PRODUCTION "${REPO}/port/src/graphics/d3d12")
set(ORIGINAL_VS "${REPO}/artifacts/shaders/dxil/2C45BFFE76D77398.vs.dxil")
foreach(REQUIRED "${ORIGINAL_VS}" "${PRODUCTION}/shaders/post_vertex_clip_gs.h"
    "${PRODUCTION}/shaders/temporal_motion_VS.h" "${PRODUCTION}/shaders/temporal_motion_PS.h")
  if(NOT EXISTS "${REQUIRED}")
    message(FATAL_ERROR "Missing existing artifact: ${REQUIRED}. This test never regenerates production headers.")
  endif()
endforeach()
add_executable(clip_motion_replay_smoke smoke.cpp
  "${PRODUCTION}/post_vertex_clip_capture.cpp" "${PRODUCTION}/temporal_motion.cpp")
target_include_directories(clip_motion_replay_smoke PRIVATE "${PRODUCTION}")
target_compile_definitions(clip_motion_replay_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(clip_motion_replay_smoke PRIVATE d3d12 dxgi dxguid)
if(MSVC)
  target_compile_options(clip_motion_replay_smoke PRIVATE /W4 /WX /EHsc)
endif()
enable_testing()
add_test(NAME clip_motion_replay_gpu COMMAND clip_motion_replay_smoke "${ORIGINAL_VS}")
set_tests_properties(clip_motion_replay_gpu PROPERTIES TIMEOUT 120)
